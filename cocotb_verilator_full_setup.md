# ============================================================
# Cocotb + Verilator Setup (Ubuntu / WSL)
# ============================================================

# ------------------------------------------------------------
# 1. Check Existing Verilator Installation
# ------------------------------------------------------------
```bash
dpkg -l | grep verilator

apt list --installed | grep verilator
```
# ------------------------------------------------------------
# 2. Check PATH
# ------------------------------------------------------------
```bash
echo $PATH
```
# ------------------------------------------------------------
# 3. Install Required Dependencies
# ------------------------------------------------------------
```bash
sudo apt update

sudo apt install -y \
    git \
    make \
    autoconf \
    g++ \
    flex \
    bison \
    ccache \
    libgoogle-perftools-dev \
    numactl \
    perl \
    python3 \
    help2man \
    python3.11 \
    python3.11-venv \
    python3.11-dev
```
# ------------------------------------------------------------
# 4. Clone Verilator
# ------------------------------------------------------------
```bash
cd ~

git clone https://github.com/verilator/verilator

cd verilator
```

# ------------------------------------------------------------
# 5. Checkout Supported Version
# ------------------------------------------------------------
```bash
git checkout v5.036
```
# ------------------------------------------------------------
# 6. Build Verilator
# ------------------------------------------------------------
```bash
autoconf

./configure

make -j$(nproc)
```
# ------------------------------------------------------------
# 7. Install Verilator
# ------------------------------------------------------------
```bash
sudo make install
```
# ------------------------------------------------------------
# 8. Verify Verilator
# ------------------------------------------------------------
```bash
verilator --version
```
# ------------------------------------------------------------
# 9. Create Python Virtual Environment
# ------------------------------------------------------------
```bash
cd ~

python3.11 -m venv cocotb311
```
# ------------------------------------------------------------
# 10. Activate Virtual Environment
# ------------------------------------------------------------
```bash
source ~/cocotb311/bin/activate
```
# ------------------------------------------------------------
# 11. Verify Python
# ------------------------------------------------------------
```bash
which python

python --version
```
# ------------------------------------------------------------
# 12. Upgrade pip Tools
# ------------------------------------------------------------
```bash
pip install --upgrade pip setuptools wheel
```
# ------------------------------------------------------------
# 13. Install Cocotb Packages
# ------------------------------------------------------------
```bash
pip install cocotb cocotb-bus cocotb-test cocotbext-axi
```
# ------------------------------------------------------------
# 14. Verify Cocotb Installation
## -----------------------------------------------------------
```bash
python -c "import cocotb; print(cocotb.__version__)"
```
### -----------------------------------------------------------
## Others Commands . 
#### Find Existing Virtual Environments
### -----------------------------------------------------------
```bash
find ~ -type f -name "activate" 2>/dev/null
```
## -----------------------------------------------------------
# 16. Go To RTL Project
## -----------------------------------------------------------
```bash
cd <PATH\>smarttimer/rtl
```
# ------------------------------------------------------------
# 17. Clean Previous Builds
# ------------------------------------------------------------
```bash
make clean

rm -rf sim_build
```
# ------------------------------------------------------------
# 18. Run Cocotb Simulation
# ------------------------------------------------------------
```bash
make SIM=verilator
```
